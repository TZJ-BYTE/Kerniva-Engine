const idleDiagnosticDBs = 16

func TestIdleDiagnosticSource(t *testing.T) {
	root := os.Getenv("IDLE_DIAGNOSTIC_SOURCE_ROOT")
	if root == "" {
		t.Skip("set IDLE_DIAGNOSTIC_SOURCE_ROOT to a converged source copy")
	}
	handler, lease, err := OpenLeasedSource(root)
	if err != nil {
		t.Fatal(err)
	}
	defer lease.Close()
	listener, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	var connections, pulls atomic.Uint64
	server := &http.Server{
		ReadHeaderTimeout: 5 * time.Second,
		ConnState: func(_ net.Conn, state http.ConnState) {
			if state == http.StateNew {
				connections.Add(1)
			}
		},
		Handler: http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			if r.URL.Path == "/pull" {
				pulls.Add(1)
			}
			handler.ServeHTTP(w, r)
		}),
	}
	ctx, stop := signal.NotifyContext(context.Background(), syscall.SIGTERM)
	defer stop()
	go func() {
		<-ctx.Done()
		server.Close()
	}()
	fmt.Printf("source_url=http://%s\n", listener.Addr())
	if err := server.Serve(listener); err != http.ErrServerClosed {
		t.Fatal(err)
	}
	fmt.Printf("source_pulls=%d source_new_connections=%d\n", pulls.Load(), connections.Load())
}

type idleDiagnosticEpoch struct {
	arrived chan struct{}
	release chan struct{}
}

func newIdleDiagnosticEpoch() *idleDiagnosticEpoch {
	return &idleDiagnosticEpoch{arrived: make(chan struct{}, idleDiagnosticDBs), release: make(chan struct{})}
}

func (epoch *idleDiagnosticEpoch) wait(t *testing.T) {
	t.Helper()
	deadline := time.NewTimer(15 * time.Second)
	defer deadline.Stop()
	for range idleDiagnosticDBs {
		select {
		case <-epoch.arrived:
		case <-deadline.C:
			t.Fatal("timed out parking all database workers")
		}
	}
}

func TestIdleDiagnosticFollower(t *testing.T) {
	root := os.Getenv("IDLE_DIAGNOSTIC_FOLLOWER_ROOT")
	if root == "" {
		t.Skip("set IDLE_DIAGNOSTIC_FOLLOWER_ROOT to a converged follower copy")
	}
	interval, err := time.ParseDuration(os.Getenv("IDLE_DIAGNOSTIC_INTERVAL"))
	if err != nil {
		t.Fatal(err)
	}
	duration, err := time.ParseDuration(os.Getenv("IDLE_DIAGNOSTIC_DURATION"))
	if err != nil {
		t.Fatal(err)
	}
	profile := os.Getenv("IDLE_DIAGNOSTIC_PROFILE_PREFIX")
	if profile != "" {
		rate, err := strconv.Atoi(os.Getenv("IDLE_DIAGNOSTIC_PROFILE_RATE"))
		if err != nil {
			t.Fatal(err)
		}
		runtime.MemProfileRate = rate
	}
	var databases [idleDiagnosticDBs]int
	for db := range databases {
		databases[db] = db
	}

	follower, err := OpenFollowerWithOptions(root, os.Getenv("IDLE_DIAGNOSTIC_UPSTREAM"), databases[:], FollowerOptions{PullIdleTimeout: 40 * time.Second})
	if err != nil {
		t.Fatal(err)
	}
	defer follower.Close()

	var cursors [idleDiagnosticDBs][]byte
	var statuses [idleDiagnosticDBs]Status
	for db := range databases {
		statuses[db] = follower.byNumber[db].result.Load().status
		if !statuses[db].Ready || statuses[db].LastError != "" {
			t.Fatalf("db %d initial status: %+v", db, statuses[db])
		}
		cursors[db], err = os.ReadFile(filepath.Join(root, "follower-v1", dbName(db), "CURRENT"))
		if err != nil {
			t.Fatal(err)
		}
	}

	var counts [idleDiagnosticDBs]uint64
	var failures [idleDiagnosticDBs]error
	var changed [idleDiagnosticDBs]uint64
	idleDiagnosticPollResult = func(db int, noChange bool, err error) {
		switch {
		case err != nil:
			failures[db] = err
		case noChange:
			counts[db]++
		default:
			changed[db]++
		}
	}
	var choice atomic.Pointer[idleDiagnosticEpoch]
	idleDiagnosticBeforePoll = func(ctx context.Context, db int) {
		if epoch := choice.Load(); epoch != nil && counts[db] >= 3 {
			epoch.arrived <- struct{}{}
			select {
			case <-epoch.release:
			case <-ctx.Done():
			}
		}
	}

	ctx, stop := context.WithCancel(context.Background())
	defer stop()
	begin, end := newIdleDiagnosticEpoch(), newIdleDiagnosticEpoch()
	choice.Store(begin)
	done := make(chan error, 1)
	go func() { done <- follower.Run(ctx, interval) }()
	begin.wait(t)
	baseline := counts
	timer := time.NewTimer(duration)
	defer timer.Stop()
	timer.Stop()

	if profile != "" {
		writeIdleDiagnosticProfile(t, profile+".warm.pprof")
		writeIdleDiagnosticProfile(t, profile+".before.pprof")
	}
	lastBefore := time.Now()
	var first, last runtime.MemStats
	runtime.ReadMemStats(&first)
	choice.Store(nil)
	timer.Reset(duration)
	close(begin.release)
	<-timer.C
	choice.Store(end)
	end.wait(t)
	runtime.ReadMemStats(&last)
	elapsed := time.Since(lastBefore)
	finished := counts
	if profile != "" {
		writeIdleDiagnosticProfile(t, profile+".after.pprof")
	}

	for db := range databases {
		if failures[db] != nil || changed[db] != 0 {
			t.Fatalf("db %d poll error=%v, changed=%d", db, failures[db], changed[db])
		}
		if current := follower.byNumber[db].result.Load().status; current != statuses[db] {
			t.Fatalf("db %d status changed from %+v to %+v", db, statuses[db], current)
		}
		current, err := os.ReadFile(filepath.Join(root, "follower-v1", dbName(db), "CURRENT"))
		if err != nil || !bytes.Equal(current, cursors[db]) {
			t.Fatalf("db %d CURRENT changed or unreadable: %v", db, err)
		}
	}
	stop()
	if err := <-done; err != nil {
		t.Fatal(err)
	}
	idleDiagnosticPollResult = nil
	idleDiagnosticBeforePoll = nil

	var total, warmup uint64
	var completed [idleDiagnosticDBs]uint64
	for db := range databases {
		completed[db] = finished[db] - baseline[db]
		if completed[db] == 0 {
			t.Fatalf("db %d completed zero measured polls", db)
		}
		total += completed[db]
		warmup += baseline[db]
	}
	result, err := json.Marshal(map[string]any{
		"interval": interval.String(), "elapsed_seconds": elapsed.Seconds(),
		"completed_no_change": total, "no_change_per_db": completed, "warmup_no_change": warmup,
		"total_alloc_bytes":      last.TotalAlloc - first.TotalAlloc,
		"alloc_bytes_per_pull":   float64(last.TotalAlloc-first.TotalAlloc) / float64(total),
		"alloc_objects_per_pull": float64(last.Mallocs-first.Mallocs) / float64(total),
		"automatic_gc_cycles":    last.NumGC - first.NumGC,
	})
	if err != nil {
		t.Fatal(err)
	}
	fmt.Println(string(result))
}

func writeIdleDiagnosticProfile(t *testing.T, path string) {
	t.Helper()
	// Complete two mark/sweep accounting cycles while polling workers are parked.
	runtime.GC()
	runtime.GC()
	file, err := os.Create(path)
	if err != nil {
		t.Fatal(err)
	}
	if err := pprof.Lookup("allocs").WriteTo(file, 0); err != nil {
		file.Close()
		t.Fatal(err)
	}
	if err := file.Close(); err != nil {
		t.Fatal(err)
	}
}
